until game.stage == 18 and game.room == 0 max 300
poke Enemies_god 1
poke vk_enemies+162 1

poke vk_game+16 1 32
poke vk_game+22 18
poke vk_game+20 8
poke vk_front_mode 4 16

poke vk_game+1 120

wait 60
poke Enemies_god 1
poke vk_enemies+162 0
expect ram8('vk_gfx16') == 1
# Premiere forme : la boule de feu (type 5) est une sphere rouge 16 bits en PAL2 (tuile 760 + 44)
until ram8('dracula+224') == 5 max 1500
wait 2
shot zzze-dracula-boule
set pal2 lambda: sorted({(ram16('vdpSpriteCache+%d' % (8 * i + 4)) >> 13) & 3 for i in range(80) if 760 <= (ram16('vdpSpriteCache+%d' % (8 * i + 4)) & 0x7FF) < 856})
expect pal2() == [2] # les cellules de Dracula (tuiles 760 a 855) sont toutes en PAL2
set face sorted({c for c in colors(80, 60, 180, 170)})
expect (238, 206, 139) in face and (172, 101, 65) in face and (205, 137, 139) not in face # peau du visage de fond : rampe chaude 8B (vk_dracula16_face_palette), plus la rampe rose MSX
set ball linked_sprites(804, 807)
expect len(ball) >= 1
set bc sorted({c for c in colors(ball[0][1], ball[0][2], ball[0][1] + 16, ball[0][2] + 16)})
expect (172, 32, 32) in bc and (139, 0, 0) in bc # sphere rouge de Castlevania-MD
# Le corps ouvert (etat 5, couches de buste 3 et 4) : tete, buste (cape rouge) et cape du bas, avec leurs tuiles 16 bits
until ram8('dracula+1') == 5 max 3000
wait 6
shot zzze-dracula-corps
expect ram8('dracula+11') == 94 and ram8('dracula+686') == 2 # pose 5E, couches 3 et 4
set bx ram8('dracula+679')
set by ram8('dracula+3')
set cloak sorted({c for c in colors(bx - 16, 145, bx + 16, 177)})
expect (139, 0, 0) in cloak and (172, 32, 32) in cloak and (65, 0, 0) in cloak # reds of the cloak (bust, 16-bit palette)
expect (252, 0, 0) not in cloak and (238, 0, 0) not in cloak # not the red of the MSX palette
set head sorted({c for c in colors(ram8('dracula+5') - 8, by - 62, ram8('dracula+5') + 8, by - 46)})
expect (172, 101, 238) in head # the head of Castlevania-MD (violet, exact entry of the retouched palette)
set cape sorted({c for c in colors(ram8('dracula+5') - 16, by - 14, ram8('dracula+5') + 16, by)})
expect (172, 32, 32) in cape and (139, 0, 0) in cape
expect len(linked_sprites(760, 855)) >= 3 # tete, 2 cellules de cape (et la cellule de capuchon vide)
expect pal2() == [2]
# Seconde forme : chauves-souris et faucilles dessinees en 16 bits (PAL2)
repeat 500 zzp_dracula_hit1.inc
repeat 700 zzp_dracula_pin.inc
expect ram8('dracula') == 23
until len(linked_sprites(760, 855)) >= 4 max 1500
wait 3
shot zzze-dracula-forme2
expect pal2() == [2]
set sc sorted({c for s in linked_sprites(760, 855) for c in colors(s[1], s[2], s[1] + 16, s[2] + 16)})
echo sorted(sc)
expect (172, 170, 238) in sc and (238, 238, 238) in sc # steel sickles (blade and rim, touch-up 3A) and bats in the violet-blue of the 16-bit palette
# Mort : les flammes (types 12 et 30, motifs F4 et F8) en 16 bits, orange de la palette fixe (captures a chaque image ou une flamme est a l'ecran)
set fl []
repeat 500 zzze_dracula_hit2_flames.inc
echo fl
expect len(fl) > 0 # des flammes ont ete a l'ecran
expect (238, 170, 0) in fl # orange de la flamme
expect pal2() == [2]

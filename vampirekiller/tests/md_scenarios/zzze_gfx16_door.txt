# (zzze) Mode graphique 16 bits (research/gfx16.md, section "Barreau de porte 16 bits") : le barreau de la porte de la salle 7 du hall
# (item_view.c draw_door, 16 x 48 pixels) est une barre de fer 16 bits (contour noir, acier, trois colliers rives) de la disposition du
# barreau MSX (vk_door_bar_rows), dessinee avec la meme ligne de palette (PAL2) et les seules entrees 2, 13, 12 et 14, fixes. Meme
# parcours que zzv_gfx16_key_door (cle de la salle 2, porte de la salle 7), puis le barreau ferme, leve a mi-hauteur et ouvert.
include boot_to_courtyard.inc
poke vk_gfx16 1                                    # 16-bit graphics from the first castle room on
until front_mode == MODES['INTERIOR'] max 3000 RIGHT
expect ram8('vk_gfx16_on') == 1 # hall room 0 (scene 6) drawn with the 16-bit cells
shot zzze-porte-hall
# Changement de salle debug du hall : A tenu, puis A+DROITE (le premier front est perdu).
poke vk_debug_keys 1                               # debug shortcuts are off by default
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
expect game.room == 2 # salle 2 du hall
poke vk_game+0 166
poke vk_game+1 192
wait 4
expect ram8('items+4') == 0 # cle cachee dans le bloc (bougie C490, mode 3)
# Regard a gauche puis fouet : X fouet 164 - 20 = 144, |144 - 136| < 20 (8450).
hold LEFT 2
hold B 2
until ram8('items+4') == 1 max 30
wait 10
shot zzze-porte-cle
until ram8('items') & 1 max 200 LEFT
expect ram8('items') == 1 and ram8('items+1') == 1 # cle blanche prise (C701 bit 0)
expect 140 <= game.player.x <= 152 # ramassage au contact (K1 : x 148)
wait 4
shot zzze-porte-cle-prise
# Salle 7, sur le rebord y 96 a gauche de la porte.
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
expect game.room == 7 # salle 7 du hall
# La panthere de la salle 7 (C0,60) nait sur le rebord de la porte ; on la retire et on
# laisse passer un eventuel recul pour isoler la porte (le run K3 du MSX encaissait son coup).
until enemies.enemy[0].active max 40
poke enemy_state+0 0
wait 30
poke vk_game+0 200
poke vk_game+1 96
wait 4
shot zzze-porte-fermee
# Barreau ferme (x 232..247, y 48..95) : acier 16 bits des entrees 2, 13, 12 et 14 de la salle (222, 444, AAA, EEE), plus le rose MSX (64C) ni
# son gris.
set dc sorted({c for c in colors(232, 48, 248, 96)})
echo dc
expect (32, 32, 32) in dc and (65, 68, 65) in dc and (172, 170, 172) in dc and (238, 238, 238) in dc
expect (205, 68, 98) not in dc # le rose MSX (entree 3, 64C)
set closed [(x, y) for y in range(48, 96) for x in range(232, 248) if pixel(x, y) != (0, 0, 0)]
expect len(closed) > 150 # le barreau garde sa taille (MSX : 4 colonnes de fut et trois colliers de 6 x 5)
expect min(x for x, y in closed) >= 232 and max(x for x, y in closed) < 248 and min(y for x, y in closed) >= 48 and max(y for x, y in closed) < 96
until ram8('items+2') == 1 max 120 RIGHT
expect game.player.x == 228 # ouverture a x 228, regard a droite
expect ram8('items') == 0 # cle consommee
wait 2
shot zzze-porte-levee-debut
# La levee : le barreau monte (draw_door a chaque tour), il reste de l'acier 16 bits a chaque image de la levee, puis il a disparu.
until ram8('items+2') == 2 max 200 RIGHT
shot zzze-porte-levee
wait 60
# Ouvert : le barreau est monte (comme en MSX il reste son dernier collier, au bord haut du cadre : rangees 48 a 51), il n'occupe
# plus que ces quelques rangees (la zone du bas est celle du rebord et de Simon).
set opened [(x, y) for y in range(48, 64) for x in range(232, 248) if pixel(x, y) != (0, 0, 0)]
set closed64 [(x, y) for (x, y) in closed if y < 64]
expect len(opened) * 2 < len(closed64) and max(y for x, y in opened) < 56

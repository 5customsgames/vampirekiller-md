# (zzze7) Mode graphique 16 bits (research/gfx16.md, section "Ennemis 16 bits (bloc 7)") : les ennemis communs du stage 18
# (hors boss final et objet 1B) sont dessines avec les sprites 16 bits de Castlevania-MD, charges a la place des cellules du
# pool tardif : chevalier a la hache (10) et sa hache aux salles 1, 2 et 8, puce (0D) aux salles 3 a 7. Deux palettes PAL2
# (salles de la puce, salles du chevalier) ; les entrees fixes gardent les couleurs MSX du stage 18.
until front_mode == MODES['TITLE'] max 2000
wait 2
poke vk_gfx16 1
tap START
tap START
until front_mode == MODES['COURTYARD'] max 500
wait 60
tap START
until ram8('paused') == 1 max 30
tap UP
tap UP
tap DOWN
tap DOWN
tap LEFT
tap RIGHT
tap LEFT
tap RIGHT
tap B
tap A
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap DOWN
tap B
expect ram8('vk_gfx16_on') == 1
until game.stage == 18 and game.room == 0 max 300
poke Enemies_god 1
until game.player.mode == 0 max 300
wait 20
expect pixel(200, 11) == (238, 238, 238) and pixel(135, 11) == (238, 0, 32)
include zzp_next.inc
until game.stage == 18 and game.room == 1 max 300
poke Enemies_god 1
until game.player.mode == 0 max 300
wait 20
until any(enemies.enemy[k].active and enemies.enemy[k].type == 20 for k in range(7)) max 900
wait 10
shot zzze7-chevalier
set axx [enemies.enemy[k].x for k in range(7) if enemies.enemy[k].active and enemies.enemy[k].type == 20][0]
set axy [enemies.enemy[k].y for k in range(7) if enemies.enemy[k].active and enemies.enemy[k].type == 20][0]
set axc sorted({c for c in colors(axx + (-12), axy + (-34), axx + (12), axy + (2))})
echo axc
expect (205, 206, 205) in axc and (98, 68, 172) in axc and (172, 101, 238) in axc and (172, 170, 205) in axc # chevalier : gris clair, violet, mauve et contour lilas clair (entrees libres de la palette des salles 1, 2 et 8)
set axb sorted({c for c in colors(axx + (-8), axy + (-34), axx + (8), axy + (2))})
expect (205, 101, 32) in axb and (172, 170, 205) in axb # bouclier orange et son contour clair dans la boite de 16 px de la pose MSX
expect (205, 170, 98) not in axc # plus l'or MSX (couleur 5)
expect pixel(200, 11) == (238, 238, 238) and pixel(135, 11) == (238, 0, 32)
poke vk_game+0 110
poke vk_game+1 96
until any(enemies.objects[k].active and enemies.objects[k].type == 21 for k in range(15)) max 900
wait 2
shot zzze7-hache
set hxx [enemies.objects[k].x for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 21][0]
set hxy [enemies.objects[k].y for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 21][0]
set hxc sorted({c for c in colors(hxx + (-12), hxy + (-18), hxx + (12), hxy + (2))})
echo hxc
expect (98, 68, 172) in hxc and (205, 206, 205) in hxc # hache violette et grise (memes entrees)
include zzp_next.inc
until game.stage == 18 and game.room == 2 max 300
poke Enemies_god 1
until game.player.mode == 0 max 300
wait 20
include zzp_next.inc
until game.stage == 18 and game.room == 3 max 300
poke Enemies_god 1
until game.player.mode == 0 max 300
wait 20
until any(enemies.enemy[k].active and enemies.enemy[k].type == 14 for k in range(7)) max 900
wait 10
shot zzze7-puce
set flx [enemies.enemy[k].x for k in range(7) if enemies.enemy[k].active and enemies.enemy[k].type == 14][0]
set fly [enemies.enemy[k].y for k in range(7) if enemies.enemy[k].active and enemies.enemy[k].type == 14][0]
set flc sorted({c for c in colors(flx + (-10), fly + (-16), flx + (10), fly + (2))})
echo flc
expect (65, 0, 0) in flc and (98, 68, 0) in flc and (172, 101, 32) in flc and (238, 170, 0) in flc # puce rouge sombre, brune et orange (entrees libres 4, 10, 6 et 11 de la palette des salles 3 a 7)
expect (238, 170, 139) in flc and (98, 32, 32) in flc # entrees fixes 1 et 13 (couleurs MSX du stage 18 gardees)
expect (172, 101, 0) not in flc # plus le brun MSX (couleur 9)
expect pixel(200, 11) == (238, 238, 238) and pixel(135, 11) == (238, 0, 32)

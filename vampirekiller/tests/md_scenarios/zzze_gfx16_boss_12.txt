# Corps 16 bits (b4_frank, b4_flea) en PAL2 : les entrees libres de vk_boss12_16_palette (864, 642) n'existent pas dans la palette MSX de la salle.
set pal lambda: [(ram16('vdpSpriteCache+%d' % (8 * i + 4)) >> 13) & 3 for i in range(80)]
expect 2 in pal()
set bc sorted({c for c in colors(0, 120, 256, 192)})
expect (65, 101, 139) in bc and (32, 68, 98) in bc
# Tir de l'homme (tuile 60 du bloc : 760 + 60 = 820), sphere rouge 16 bits, tire toutes les 48 images quand Igor est en l'air.
set sh lambda: sorted((x, y) for (tl, x, y) in linked_sprites(820, 820))
until sh() != [] max 400
set sp sh()[0]
shot zzze-boss12-tir
set sc sorted({c for c in colors(sp[0] - 8, sp[1] - 8, sp[0] + 16, sp[1] + 16)})
expect (172, 32, 32) in sc and (238, 170, 139) in sc

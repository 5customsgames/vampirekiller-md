# (zzze) Mode graphique 16 bits (research/gfx16.md, section "Bougies et torches 16 bits") : l'objet 1A (goutte bleue que
# liberent les bougies d'objet 15 des stages 4 a 6) est une goutte de gel ombree (dessinee sur la silhouette MSX) en PAL2, avec
# les seules entrees fixes (2, 14, 15) : memes cellules, memes boites, memes images de salle que le MSX. Stage 4 salle 0, comme zzi.
until front_mode == MODES['TITLE'] max 2000
wait 2
poke vk_gfx16 1
include zy_enter4.inc
poke Enemies_god 1
expect ram8('vk_gfx16_on') == 1
wait 60
expect game.stage == 4 and game.room == 0
expect ram8('vk_candles+4') & 4 # bougie (128, 64) allumee
poke vk_game+0 116
poke vk_game+1 96
poke vk_game+2 0
wait 4
hold B 1
until (ram8('vk_candles+4') & 4) == 0 max 12
set f0 frame
until any(enemies.enemy[k].active and enemies.enemy[k].type == 15 for k in range(7)) max 60
set f1 frame
expect f1 - f0 == 31 # meme delai qu'en MSX (zzi)
set g [k for k in range(7) if enemies.enemy[k].active and enemies.enemy[k].type == 15][0]
expect (enemies.enemy[g].x, enemies.enemy[g].y, enemies.enemy[g].state, enemies.enemy[g].pose) == (128, 80, 0, 0x9F)
wait 2
shot zzze-goutte1a-a
set ba sorted({c for c in colors(122, 68, 134, 80)})
echo ba
expect (0, 0, 238) in ba and (238, 238, 238) in ba # bleu de l'entree 15 et reflet de l'entree 14 (le MSX n'a que le bleu et le noir)
# La seconde pose (pose paire) : la goutte aplatie, toujours les memes entrees.
until enemies.enemy[g].state == 2 max 60
shot zzze-goutte1a-sol
set bb sorted({c for c in colors(122, 86, 134, 96)})
echo bb
expect (0, 0, 238) in bb and (238, 238, 238) in bb and (32, 32, 32) in bb

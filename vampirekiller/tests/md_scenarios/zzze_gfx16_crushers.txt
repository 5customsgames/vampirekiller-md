# (zzze) Mode graphique 16 bits (research/gfx16.md, section "Ecraseurs 16 bits") : les trois ecraseurs a pointes de la salle 1 du stage 6
# (crushers.c) sont dessines en 16 bits : chaine d'anneaux d'acier, barre rivetee et quatre pointes a bout de sang, memes cellules, memes
# boites et memes 128 tuiles de bloc que le MSX ; la salle (sans autre ennemi) a sa PAL2 : les entrees des autres contenus gardent les
# couleurs MSX, les entrees 4, 5, 6 et 9 portent l'acier et le sang. Meme parcours que zx_stage6_enemies (selecteur de pause).
until front_mode == MODES['TITLE'] max 2000
wait 2
poke vk_gfx16 1
tap START
tap START
until front_mode == MODES['COURTYARD'] max 500
wait 60
include zzze_pause.inc
repeat 5 zzze_down.inc
tap RIGHT
tap B
until game.stage == 6 and game.room == 1 max 300
expect ram8('vk_gfx16_on') == 1
poke Enemies_god 1
until game.player.mode == 0 max 300
wait 10
until ram8('vk_crushers') != 0 max 60
wait 6
shot zzze-ecraseurs-a
set cx ram8('vk_crushers+2')
set cy ram8('vk_crushers+1')
echo (cx, cy)
set ca sorted({c for c in colors(cx, cy - 4, cx + 32, cy + 12)})
echo ca
expect (205, 206, 205) in ca and (65, 68, 65) in ca and (139, 137, 139) in ca # acier : entrees 6 (clair), 4 (ombre) et 5
expect (98, 0, 0) in ca or (238, 238, 238) in ca # sang des pointes ou reflet des rivets
expect (172, 170, 172) in ca and (32, 32, 32) in ca # entrees fixes 12 et 2 (reflet, contour)
# Plus aucune couleur MSX des ecraseurs (entree 4 : 046, entree 6 : 48A, entree 11 : 444 de la salle MSX) sur la barre.
expect (98, 32, 0) not in ca and (205, 137, 98) not in ca
wait 12
shot zzze-ecraseurs-b
wait 24
shot zzze-ecraseurs-c
# Les entrees fixes sont celles de la salle : le texte et le HUD du haut gardent leurs couleurs (blanc, rouge, bleu de la barre).
expect pixel(200, 11) == (238, 238, 238) and pixel(135, 11) == (238, 0, 32)
# Les ecraseurs montent et descendent comme avant : meme enregistrement, la pose seule change.
set y0 ram8('vk_crushers+1')
wait 30
expect ram8('vk_crushers+1') != y0

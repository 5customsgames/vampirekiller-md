# (zzze) Mode graphique 16 bits (research/gfx16.md, section "Ennemis 16 bits (bloc 2)") : les ennemis communs du bloc 2
# (stages 4 a 6) sont dessines avec les sprites 16 bits de Castlevania-MD : chevalier (objet 06, salle 0 du stage 4),
# tete de meduse (objet 08, salle 4), chevalier du stage 5 (salle 3), chauve-souris (objet 07, salle 0 du stage 6), statue
# (objet 0A, colonne d'os) et sa boule (salle 2). Les pics de la salle 1 du stage 6 et l'objet 1A restent en MSX.
until front_mode == MODES['TITLE'] max 2000
wait 2
poke vk_gfx16 1
include zy_enter4.inc
poke Enemies_god 1
expect ram8('vk_gfx16_on') == 1
wait 60
shot zzze2-chevaliers
# Les cases du HUD gardent leurs couleurs, comme au bloc 1.
expect pixel(200, 11) == (238, 238, 238) # case de droite, haut
expect pixel(135, 11) == (238, 0, 32) # case rouge, haut
set kx [enemies.objects[k].x for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 6][0]
set ky [enemies.objects[k].y for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 6][0]
set kc sorted({c for c in colors(kx - 8, ky - 31, kx + 8, ky + 1)})
expect (172, 101, 238) in kc # violet 16 bits (entree libre 5 de PAL2)
expect (205, 170, 98) not in kc # plus l'orange MSX (entree 5 de la palette MSX)
# Salle 4 : tete de meduse
poke vk_game+20 3
poke vk_debug_keys 1
hold A 3
hold A+RIGHT 3
wait 2
expect game.room == 4
until any(enemies.objects[k].active and enemies.objects[k].type == 7 for k in range(15)) max 400
wait 8
shot zzze2-meduse
set mx [enemies.objects[k].x for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 7][0]
set my [enemies.objects[k].y for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 7][0]
set mc sorted({c for c in colors(mx - 8, my - 14, mx + 8, my + 2)})
expect (32, 68, 139) in mc # bleu des cheveux 16 bits (entree libre 10)
expect (205, 170, 98) not in mc
# Stage 5, salle 3 : chevaliers
poke vk_game+22 5
poke vk_game+20 2
hold A 3
hold A+RIGHT 3
wait 20
expect game.stage == 5 and game.room == 3
until any(enemies.objects[k].active and enemies.objects[k].type == 6 for k in range(15)) max 200
wait 30
shot zzze2-stage5
set kx [enemies.objects[k].x for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 6][0]
set ky [enemies.objects[k].y for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 6][0]
set kc sorted({c for c in colors(kx - 8, ky - 31, kx + 8, ky + 1)})
expect (172, 101, 238) in kc and (205, 170, 98) not in kc # chevalier 16 bits au stage 5
# Stage 6, salle 0 : chauve-souris
poke vk_game+22 6
poke vk_game+20 1
hold A 3
hold A+LEFT 3
wait 20
expect game.stage == 6 and game.room == 0
poke vk_game+0 232
poke vk_game+1 128
until any(enemies.objects[k].active and enemies.objects[k].type == 8 for k in range(15)) max 400
wait 40
shot zzze2-chauve-souris
set bx [enemies.objects[k].x for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 8][0]
set by [enemies.objects[k].y for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 8][0]
set bc sorted({c for c in colors(bx - 7, by - 14, bx + 9, by + 2)})
expect (238, 206, 238) in bc # contour lilas 16 bits (entree libre 11)
expect (238, 238, 238) not in bc # plus le blanc MSX
# Salle 2 : statue et boule
poke vk_game+20 1
hold A 3
hold A+RIGHT 3
wait 20
expect game.stage == 6 and game.room == 2
poke vk_game+0 232
poke vk_game+1 160
until any(enemies.objects[k].active and enemies.objects[k].type == 10 for k in range(15)) max 300
wait 6
shot zzze2-statue-boule
set sx [enemies.objects[k].x for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 9][0]
set sy [enemies.objects[k].y for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 9][0]
set sc sorted({c for c in colors(sx - 8, sy - 31, sx + 8, sy + 1)})
expect (172, 101, 238) in sc and (238, 238, 238) not in sc # colonne d'os violette, plus la statue blanche MSX
set fx [enemies.objects[k].x for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 10][0]
set fy [enemies.objects[k].y for k in range(15) if enemies.objects[k].active and enemies.objects[k].type == 10][0]
# Boule : sphere rouge 8 x 8 (rouge MSX de l'entree fixe 8, coeur orange de l'entree 7), la MSX est un disque rouge uni.
set fc colors(fx - 4, fy - 12, fx + 12, fy + 4)
expect len([c for c in fc if c == (238, 0, 32)]) >= 10 and (205, 101, 32) in fc
until any(enemies.objects[k].active and enemies.objects[k].type == 9 and (enemies.objects[k].extraPhase & 64) for k in range(15)) max 300
wait 3
shot zzze2-statue-yeux
set sc2 sorted({c for c in colors(sx - 8, sy - 31, sx + 8, sy + 1)})
expect sc2 != sc and (238, 206, 238) not in sc2 # yeux : l'image rouge sombre de b2_pillar_flash
expect (238, 238, 238) in sc2 and (238, 238, 238) not in sc # yeux allumes : orbites blanches (entree fixe 14), absentes de la pose normale
expect len([c for c in colors(sx - 8, sy - 31, sx + 8, sy + 1) if c == (238, 238, 238)]) >= 12 # au moins 12 pixels blancs (22 orbites, quelques-uns caches)
# Salle 1 : les pics restent en MSX, avec la palette MSX
hold A 3
hold A+LEFT 3
wait 2
expect game.stage == 6 and game.room == 1
poke vk_game+0 196
poke vk_game+1 160
wait 60
shot zzze2-pics
# Les pics (ecraseurs) sont maintenant en 16 bits (zzze_gfx16_crushers) : acier des entrees 6 et 4 de la palette de la salle (CCC et 444),
# plus le dore MSX (046 et 48A).
set pc sorted({c for c in colors(0, 32, 256, 120)})
expect (205, 206, 205) in pc and (65, 68, 65) in pc and (98, 68, 0) not in pc and (172, 137, 65) not in pc

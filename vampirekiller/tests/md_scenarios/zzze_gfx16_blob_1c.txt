# (zzze) Mode graphique 16 bits (research/gfx16.md, section "Bougies et torches 16 bits") : l'objet 1C (goutte blanche que
# liberent les bougies d'objet 15 des stages 7 a 9) est une goutte de gel ombree (blanc, gris clair, gris sombre, contour noir)
# dessinee sur la silhouette MSX, en PAL2 avec les seules entrees fixes (2, 12, 13, 14) : memes cellules (B4 et B8), memes boites.
# Stage 9 salle 2, comme zzf (bougie (48, 64), objet 1C cree en (48, 80)).
until front_mode == MODES['TITLE'] max 2000
wait 2
poke vk_gfx16 1
include zzc_enter9.inc
poke Enemies_god 1
expect ram8('vk_gfx16_on') == 1
wait 60
include zzb_next.inc
until game.stage == 9 and game.room == 1 max 200
include zzb_next.inc
until game.stage == 9 and game.room == 2 max 200
wait 20
expect ram8('vk_candles+4') & 1 # bougie (48, 64) allumee
poke vk_game+0 36
poke vk_game+1 112
wait 4
tap C 1
until game.player.y <= 98 max 30
hold B 1
until (ram8('vk_candles+4') & 1) == 0 max 12
set f0 frame
until any(enemies.enemy[k].active and enemies.enemy[k].type == 15 for k in range(7)) max 60
set f1 frame
expect f1 - f0 == 31 # meme delai qu'en MSX (zzf : 843 -> 874)
set g [k for k in range(7) if enemies.enemy[k].active and enemies.enemy[k].type == 15][0]
expect (enemies.enemy[g].x, enemies.enemy[g].y, enemies.enemy[g].state, enemies.enemy[g].pose) == (48, 80, 0, 0x9F)
wait 2
shot zzze-goutte1c-a
set ca sorted({c for c in colors(42, 68, 54, 80)})
echo ca
expect (238, 238, 238) in ca and (172, 170, 172) in ca # blanc (entree 14) et gris clair (entree 12) ; le MSX n'a que le blanc et le noir
until enemies.enemy[g].state == 2 max 60
shot zzze-goutte1c-sol
set cb sorted({c for c in colors(42, 100, 54, 112)})
echo cb
expect (238, 238, 238) in cb and (0, 0, 0) in cb
